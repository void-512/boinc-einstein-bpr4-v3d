cmake_minimum_required(VERSION 3.20)
project(brp_v3d VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Vulkan REQUIRED)
find_program(GLSLC glslc REQUIRED)
find_library(FFTW3F NAMES fftw3f libfftw3f.so.3 REQUIRED)
set(REFERENCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/boinc-app-eah-brp" CACHE PATH "Historical BRP sources, read only")
set_source_files_properties("${REFERENCE_DIR}/hs_common.c" "${REFERENCE_DIR}/demod_binary_resamp_cpu.c" "${REFERENCE_DIR}/demod_binary_fft_fftw.c" PROPERTIES LANGUAGE CXX)
add_library(brp_science src/science.cpp src/reference_adapter.cpp "${REFERENCE_DIR}/hs_common.c" "${REFERENCE_DIR}/demod_binary_resamp_cpu.c" "${REFERENCE_DIR}/demod_binary_fft_fftw.c" "${REFERENCE_DIR}/erp_utilities.cpp")
set_source_files_properties("${REFERENCE_DIR}/erp_utilities.cpp" PROPERTIES COMPILE_DEFINITIONS LOGLEVEL=error)
target_include_directories(brp_science PUBLIC include PRIVATE "${REFERENCE_DIR}" tests/shims)
target_link_libraries(brp_science PUBLIC "${FFTW3F}")
target_compile_options(brp_science PRIVATE -Wall -Wextra -ffp-contract=off)
set(SHADERS resample mean prepare fft fft_local power harmonic fft_stockham fft_radix4)
option(BRP_DIVISION_FFT "Reproduce the initial FFT integer-division baseline" OFF)
option(BRP_SCATTER_PREPARE "Reproduce the initial bit-reversal scattered writes" OFF)
option(BRP_STRIDED_TWIDDLES "Use the original strided twiddle table" OFF)
if(BRP_DIVISION_FFT)
  set(BRP_STRIDED_TWIDDLES ON)
endif()
foreach(NAME IN LISTS SHADERS)
  set(SHADER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${NAME}.comp")
  if((BRP_DIVISION_FFT AND (NAME STREQUAL "fft" OR NAME STREQUAL "fft_local")) OR (BRP_SCATTER_PREPARE AND NAME STREQUAL "prepare"))
    set(SHADER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/experiments/baseline/${NAME}.comp")
  endif()
  set(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shaders/${NAME}.spv")
  add_custom_command(OUTPUT "${OUTPUT}" COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
    COMMAND "${GLSLC}" --target-env=vulkan1.3 -O -DBRP_STRIDED_TWIDDLES=$<BOOL:${BRP_STRIDED_TWIDDLES}> -I "${CMAKE_CURRENT_SOURCE_DIR}/shaders" "${SHADER_SOURCE}" -o "${OUTPUT}"
    DEPENDS "${SHADER_SOURCE}" shaders/common.glsl VERBATIM)
  list(APPEND SPIRV "${OUTPUT}")
endforeach()
add_custom_target(brp_shaders ALL DEPENDS ${SPIRV})
add_library(brp_vulkan src/vulkan.cpp)
target_include_directories(brp_vulkan PUBLIC include)
target_link_libraries(brp_vulkan PUBLIC Vulkan::Vulkan brp_science)
target_compile_options(brp_vulkan PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
target_compile_definitions(brp_vulkan PRIVATE BRP_SHADER_DIR="${CMAKE_CURRENT_BINARY_DIR}/shaders")
target_compile_definitions(brp_vulkan PUBLIC BRP_STRIDED_TWIDDLES=$<BOOL:${BRP_STRIDED_TWIDDLES}>)
target_compile_definitions(brp_vulkan PRIVATE BRP_SCATTER_PREPARE=$<BOOL:${BRP_SCATTER_PREPARE}>)
add_dependencies(brp_vulkan brp_shaders)
add_executable(brp_bench src/main.cpp)
target_link_libraries(brp_bench PRIVATE brp_vulkan)
target_compile_options(brp_bench PRIVATE -Wall -Wextra)
enable_testing()
add_executable(brp_reference_tests tests/reference.cpp)
target_link_libraries(brp_reference_tests PRIVATE brp_science)
add_test(NAME reference COMMAND brp_reference_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME boinc_cleanup COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/exclusive_gpu_test.py")
endif()
